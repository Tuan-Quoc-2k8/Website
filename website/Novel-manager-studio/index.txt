  const keys = ['folders','chapters','characters','events','locations','organizations','countries','worldRules','items','relationships','graphPositions','books'];

  });
}

function uid() { return Date.now().toString(36) + Math.random().toString(36).slice(2, 7); }
function esc(s) { return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }
function dateStr(ts) {
  if (!ts) return '—';
  const d = new Date(ts);
  return d.toLocaleDateString('vi-VN') + ' ' + d.toLocaleTimeString('vi-VN', {hour:'2-digit',minute:'2-digit'});
}

function wordCount(text) {
  const cleaned = text.replace(/<[^>]*>/g, ' ').replace(/\s+/g, ' ').trim();
  if (!cleaned) return 0;
  // Vietnamese word count: count syllables/words separated by spaces
  return cleaned.split(' ').filter(w => w.length > 0).length;
}

function readingTime(words) {
  const mins = Math.ceil(words / 200);
  if (mins < 1) return '< 1 phút';
  return mins + ' phút';
}

function getEntityList(cat) {
  if (cat === 'characters') return D.characters;
  if (cat === 'events') return D.events;
  if (cat === 'locations') return D.locations;
  if (cat === 'organizations') return D.organizations;
  if (cat === 'countries') return D.countries;
  if (cat === 'worldRules') return D.worldRules;
  if (cat === 'items') return D.items;
  return [];
}

function getEntityName(cat, id) {
  const list = getEntityList(cat);
  const e = list.find(x => x.id === id);
  return e ? (e.name || e.rule || '—') : '—';
}

function renderFolderTree() {
  const tree = document.getElementById('folder-tree');
  if (!D.folders.length) {
    tree.innerHTML = '<div class="empty-state" style="padding:20px;text-align:center"><div style="color:var(--tx3);font-size:12px">Chưa có thư mục nào<br><br><button class="btn btn-sm btn-primary" onclick="createFolder(null)">Tạo thư mục</button></div></div>';
    return;
  }
  const rootFolders = D.folders.filter(f => !f.parentId);
  tree.innerHTML = rootFolders.map(f => renderFolderItem(f, 0)).join('');
}

function renderFolderItem(folder, depth) {
  const isOpen = UI.folderOpen[folder.id] !== false;
  const isActive = UI.activeFolderId === folder.id;
  const chapters = Object.values(D.chapters).filter(c => c.folderId === folder.id)
    .sort((a,b) => (a.order||0)-(b.order||0));
  const subFolders = D.folders.filter(f => f.parentId === folder.id);
  const indent = depth * 14;

  let html = `<div class="fi ${isActive?'on':''}" style="padding-left:${6+indent}px" 
    onclick="selectFolder('${folder.id}')"
    oncontextmenu="folderCtxMenu(event,'${folder.id}')">
    <span class="fold-btn ${isOpen?'open':''}" onclick="event.stopPropagation();toggleFolderOpen('${folder.id}')">▶</span>
    <span class="fi-icon">📁</span>
    <span class="fi-name">${esc(folder.name)}</span>
    <div class="fi-acts">
      <button class="ib" style="width:20px;height:20px;font-size:11px" onclick="event.stopPropagation();createChapter('${folder.id}')" title="Thêm chương">＋</button>
      <button class="ib" style="width:20px;height:20px;font-size:11px" onclick="event.stopPropagation();renameFolder('${folder.id}')" title="Đổi tên">✎</button>
      <button class="ib" style="width:20px;height:20px;font-size:11px;color:var(--er)" onclick="event.stopPropagation();deleteFolder('${folder.id}')" title="Xóa">✕</button>
    </div>
  </div>`;

  if (isOpen) {
    subFolders.forEach(sf => { html += renderFolderItem(sf, depth+1); });
    chapters.forEach(ch => {
      const isChActive = UI.activeChapterId === ch.id;
      html += `<div class="ch-item ${isChActive?'on':''}" style="padding-left:${26+indent}px"
        onclick="selectChapter('${ch.id}')"
        oncontextmenu="chapterCtxMenu(event,'${ch.id}')">
        <span class="ch-item-dot"></span>
        <span style="flex:1;overflow:hidden;text-overflow:ellipsis;white-space:nowrap">${esc(ch.title||'Chương chưa đặt tên')}</span>
        <span style="font-size:10px;color:var(--tx3);flex-shrink:0">${ch.wordCount||0}T</span>
      </div>`;
    });
  }
  return html;
}

}

function createFolder(parentId) {
  showModal('Tạo thư mục mới',
    `<div class="form-group"><label>Tên thư mục</label><input type="text" id="m-fname" placeholder="VD: Quyển 1 - Bình Minh Mới" style="width:100%"></div>`,
    [
      {label:'Hủy', cls:'btn-ghost', action:closeModal},
      {label:'Tạo', cls:'btn-primary', action:() => {
        const name = document.getElementById('m-fname').value.trim();
        if (!name) return toast('Nhập tên thư mục', 'err');
        const f = { id: uid(), name, parentId: parentId||null, order: D.folders.length };
        D.folders.push(f);
        saveData('folders');
        UI.activeFolderId = f.id;
        UI.folderOpen[f.id] = true;
        renderFolderTree();
        closeModal();
        toast('Đã tạo thư mục "' + name + '"');
      }}
    ]
  );
  setTimeout(() => document.getElementById('m-fname')?.focus(), 50);
}

function createFolderThenChapter() {

    `<div class="form-group"><label>Tên thư mục</label><input type="text" id="m-fname" value="Quyển 1" style="width:100%"></div>
     <div class="form-group"><label>Tiêu đề chương</label><input type="text" id="m-chname" value="Chương 1" style="width:100%"></div>`,
    [
      {label:'Hủy', cls:'btn-ghost', action:closeModal},
      {label:'Tạo', cls:'btn-primary', action:() => {
        const fname = document.getElementById('m-fname').value.trim() || 'Quyển 1';
        const chname = document.getElementById('m-chname').value.trim() || 'Chương 1';
        const f = { id: uid(), name: fname, parentId: null, order: D.folders.length };
        D.folders.push(f);
        const ch = { id: uid(), folderId: f.id, title: chname, content: '', wordCount: 0, createdAt: Date.now(), updatedAt: null, order: 0 };
        D.chapters[ch.id] = ch;
        saveData('folders'); saveData('chapters');
        UI.activeFolderId = f.id;
        UI.folderOpen[f.id] = true;
        selectChapter(ch.id);
        renderFolderTree();
        closeModal();
      }}
    ]
  );
}

function execCmd(cmd, val) {
  document.getElementById('editor').focus();
  document.execCommand(cmd, false, val || null);

}

function updateToolbarState() {

}

function onEditorInput() {

function onEditorKeydown(e) {
  const drop = document.getElementById('cmd-drop');
  if (drop.style.display === 'block') {
    const items = drop.querySelectorAll('.cmd-item');
    const sel = drop.querySelector('.cmd-item.sel');
    let idx = sel ? Array.from(items).indexOf(sel) : -1;
    if (e.key === 'ArrowDown') { e.preventDefault(); idx = Math.min(idx+1, items.length-1); items.forEach((x,i)=>x.classList.toggle('sel',i===idx)); items[idx]?.scrollIntoView({block:'nearest'}); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); idx = Math.max(idx-1, 0); items.forEach((x,i)=>x.classList.toggle('sel',i===idx)); items[idx]?.scrollIntoView({block:'nearest'}); }
    else if (e.key === 'Enter' || e.key === 'Tab') { e.preventDefault(); if(sel) sel.click(); else if(items[0]) items[0].click(); }
    else if (e.key === 'Escape') { hideCmdDrop(); }
  }
}

function onEditorKeyup(e) {
  if (!['ArrowDown','ArrowUp','Enter','Tab'].includes(e.key)) checkWritingCommands();
  updateToolbarState();
}

function checkWritingCommands() {
  const editor = document.getElementById('editor');
  const sel = window.getSelection();
  if (!sel.rangeCount) return;
  const range = sel.getRangeAt(0);
  const text = range.startContainer.textContent || '';
  const pos = range.startOffset;
  const before = text.slice(0, pos);
  const match = before.match(/=\[(\w+):/);
  if (!match) { hideCmdDrop(); return; }
  const type = match[1].toLowerCase();
  showCmdDropdown(type, range);
}

function showCmdDropdown(type, range) {
  const drop = document.getElementById('cmd-drop');
  let items = [];
  if (type === 'name') {
    items = D.characters.slice(0, 15).map(c => ({
      id: c.id, label: c.name, sub: IMP_LABELS[c.importance||0]||'', badge: IMP_CLASSES[c.importance||0], type: 'char'
    }));
  } else if (type === 'event') {
    items = D.events.slice(0, 10).map(e => ({ id: e.id, label: e.name, sub: e.date||'', type: 'event' }));
  } else if (type === 'location') {
    items = D.locations.slice(0, 10).map(l => ({ id: l.id, label: l.name, sub: l.region||'', type: 'loc' }));
  } else if (type === 'note') {
    items = [{ id: '_note', label: 'Thêm ghi chú', sub: 'Nhập nội dung ghi chú', type: 'note' }];
  } else if (type === 'bold') {
    items = [{ id: '_bold', label: 'Chữ đậm', sub: 'Bôi đậm văn bản hiện tại', type: 'fmt' }];
  } else if (type === 'italic') {
    items = [{ id: '_italic', label: 'Chữ nghiêng', sub: 'In nghiêng văn bản hiện tại', type: 'fmt' }];
  }
  if (!items.length && type !== 'note' && type !== 'bold' && type !== 'italic') {
    // Show available commands
    items = [
      {id:'_name',label:'=[name:]',sub:'Chèn tên nhân vật',type:'help'},
      {id:'_note',label:'=[note:]',sub:'Thêm ghi chú',type:'help'},
      {id:'_event',label:'=[event:]',sub:'Tham chiếu sự kiện',type:'help'},
      {id:'_loc',label:'=[location:]',sub:'Tham chiếu địa điểm',type:'help'},
    ];
  }

  if (!items.length) { hideCmdDrop(); return; }

  const rect = range.getBoundingClientRect();
  drop.style.display = 'block';
  drop.style.left = Math.min(rect.left, window.innerWidth - 220) + 'px';
  drop.style.top = (rect.bottom + 4) + 'px';
  drop.innerHTML = items.map((it,i) => `
    <div class="cmd-item${i===0?' sel':''}" onclick="insertCmdRef('${type}','${it.id}','${esc(it.label)}')">
      <div style="flex:1">
        <div class="cmd-item-name">${esc(it.label)}</div>
        <div class="cmd-item-sub">${esc(it.sub)}</div>
      </div>
      ${it.badge?`<span class="cmd-item-badge ${it.badge}">${esc(IMP_LABELS[['imp-0','imp-1','imp-2','imp-3','imp-4','imp-5'].indexOf(it.badge)])}</span>`:''}
    </div>`).join('');
}

function hideCmdDrop() {
  document.getElementById('cmd-drop').style.display = 'none';
}

function insertCmdRef(type, id, label) {
  hideCmdDrop();
  const editor = document.getElementById('editor');
  editor.focus();
  const sel = window.getSelection();
  if (!sel.rangeCount) return;
  const range = sel.getRangeAt(0);
  const node = range.startContainer;
  const text = node.textContent;
  const pos = range.startOffset;
  const before = text.slice(0, pos);
  const cmdMatch = before.match(/=\[\w+:/);
  if (!cmdMatch) return;
  const cmdStart = before.lastIndexOf(cmdMatch[0]);

  // Remove the command text and replace with styled ref
  const newText = text.slice(0, cmdStart) + text.slice(pos);
  node.textContent = newText;

  let span = document.createElement('span');
  if (type === 'name') {
    span.className = 'char-ref'; span.dataset.charId = id;
    span.textContent = label;
    span.title = 'Nhân vật: ' + label;
  } else if (type === 'event') {
    span.className = 'event-ref'; span.dataset.eventId = id;
    span.textContent = '⚡' + label;
  } else if (type === 'location') {
    span.className = 'loc-ref'; span.dataset.locId = id;
    span.textContent = '📍' + label;
  } else if (type === 'note') {
    span.className = 'note-span';
    span.textContent = '📝 Ghi chú';
    span.contentEditable = 'true';
  } else {
    span.textContent = label;
  }

  const newRange = document.createRange();
  newRange.setStart(node, cmdStart);
  newRange.setEnd(node, cmdStart);
  sel.removeAllRanges(); sel.addRange(newRange);

  range.insertNode(document.createTextNode('\u00A0'));
  range.insertNode(span);
  range.setStartAfter(span);
  range.setEndAfter(span);
  sel.removeAllRanges(); sel.addRange(range);

  onEditorInput();
}

function showCmdHelper(type) {
  const editor = document.getElementById('editor');
  editor.focus();
  document.execCommand('insertText', false, `=[${type}:`);
  checkWritingCommands();
}

function insertNote() {
  const text = prompt('Nội dung ghi chú:');
  if (!text) return;
  const editor = document.getElementById('editor');
  editor.focus();
  const span = document.createElement('span');
  span.className = 'note-span';
  span.textContent = '📝 ' + text;
  const sel = window.getSelection();
  if (sel.rangeCount) {
    const range = sel.getRangeAt(0);
    range.deleteContents();
    range.insertNode(span);
    range.setStartAfter(span);
    sel.removeAllRanges(); sel.addRange(range);
  } else { editor.appendChild(span); }
  onEditorInput();
}

    as.style.color = 'var(--wn)';
  }
}

function runValidation() {
  const warns = [];
  D.characters.forEach(c => {
    if (!c.age) warns.push(`Nhân vật "${c.name}" chưa có tuổi.`);
    if (!c.status) warns.push(`Nhân vật "${c.name}" chưa có trạng thái.`);
    if (!c.description) warns.push(`Nhân vật "${c.name}" chưa có mô tả.`);
  });
  Object.values(D.chapters).forEach(ch => {
    if (!ch.title || ch.title === 'Chương chưa đặt tên') warns.push(`Chương ID ${ch.id} chưa có tiêu đề.`);
    if (!ch.content || ch.content.replace(/<[^>]*>/g,'').trim().length < 10) warns.push(`Chương "${ch.title||'?'}" trống hoặc quá ngắn.`);
  });
  // check duplicate char names
  const names = D.characters.map(c => c.name.toLowerCase().trim());
  const dupes = names.filter((n, i) => names.indexOf(n) !== i);
  if (dupes.length) warns.push(`Có nhân vật trùng tên: ${[...new Set(dupes)].join(', ')}`);

  const count = warns.length;
  document.getElementById('warn-count').textContent = count;
  document.getElementById('warn-btn').style.display = count > 0 ? 'flex' : 'none';
  document.getElementById('warn-body').innerHTML = warns.slice(0,20).map(w =>
    `<div class="warn-item"><span class="warn-icon">⚠</span><span>${esc(w)}</span></div>`
  ).join('') || '<div class="warn-item" style="color:var(--ok)">✓ Không có cảnh báo nào</div>';
}

function toggleWarnings() {
  const p = document.getElementById('warn-panel');
  p.style.display = p.style.display === 'block' ? 'none' : 'block';
  runValidation();
}

function renderEntityList() {
  const list = document.getElementById('ent-list');
  const q = (document.getElementById('ent-search')?.value || '').toLowerCase();
  let items = getEntityList(UI.activeCategory);
  if (q) items = items.filter(x => (x.name||x.rule||'').toLowerCase().includes(q));

  if (!items.length) {
    list.innerHTML = `<div class="empty-state" style="padding:20px"><div class="text-muted text-sm">Chưa có ${CAT_LABELS[UI.activeCategory]} nào</div></div>`;
    return;
  }

  list.innerHTML = items.map(x => {
    const name = x.name || x.rule || '—';
    const isOn = UI.activeEntityId === x.id;
    let sub = '';
    if (UI.activeCategory === 'characters') sub = IMP_LABELS[x.importance||0];
    else if (UI.activeCategory === 'events') sub = x.date || '';
    else if (UI.activeCategory === 'locations') sub = x.region || '';
    else if (UI.activeCategory === 'worldRules') sub = x.category || '';

    return `<div class="ent-item ${isOn?'on':''}" onclick="selectEntity('${x.id}')">
      <span class="ent-dot"></span>
      <div style="flex:1;min-width:0">
        <div style="overflow:hidden;text-overflow:ellipsis;white-space:nowrap;font-size:12.5px;color:${isOn?'var(--ac)':'var(--tx2)'}">${esc(name)}</div>
        ${sub?`<div class="text-sm text-muted" style="overflow:hidden;text-overflow:ellipsis;white-space:nowrap">${esc(sub)}</div>`:''}
      </div>
    </div>`;
  }).join('');

let graphSVG, graphNodes, graphEdges, graphW, graphH;

function initGraph() {
  graphSVG = document.getElementById('graph-svg');
  graphNodes = document.getElementById('graph-nodes');
  graphEdges = document.getElementById('graph-edges');
  const rect = graphSVG.getBoundingClientRect();
  graphW = rect.width || 600; graphH = rect.height || 400;
}

function renderGraph() {
  if (!graphSVG) initGraph();
  if (UI.activeCategory !== 'characters') { graphNodes.innerHTML = ''; graphEdges.innerHTML = ''; return; }
  const chars = D.characters;
  if (!chars.length) {
    graphNodes.innerHTML = ''; graphEdges.innerHTML = '';
    document.getElementById('graph-hint').style.display = 'block';
    return;
  }
  document.getElementById('graph-hint').style.display = chars.length < 2 ? 'block' : 'none';

  // Assign default positions for new nodes
  chars.forEach((c, i) => {
    if (!D.graphPositions[c.id]) {
      const angle = (i / chars.length) * 2 * Math.PI;
      const r = Math.min(graphW, graphH) * 0.3;
      D.graphPositions[c.id] = {
        x: graphW/2 + r * Math.cos(angle),
        y: graphH/2 + r * Math.sin(angle)
      };
    }
  });

  // Draw edges
  graphEdges.innerHTML = D.relationships.map(r => {
    const p1 = D.graphPositions[r.source];
    const p2 = D.graphPositions[r.target];
    if (!p1 || !p2) return '';
    const mx = (p1.x + p2.x) / 2;
    const my = (p1.y + p2.y) / 2;
    return `<g class="g-edge" onclick="selectRelationship('${r.id}')">
      <line x1="${p1.x}" y1="${p1.y}" x2="${p2.x}" y2="${p2.y}" stroke="${UI.activeEntityId===r.source||UI.activeEntityId===r.target?'#c8a96e':'#363650'}" stroke-width="${UI.activeEntityId===r.source||UI.activeEntityId===r.target?'2':'1.5'}"/>
      <text class="edge-label" x="${mx}" y="${my-5}">${esc(r.type||'')}</text>
    </g>`;
  }).join('');

  // Draw nodes
  graphNodes.innerHTML = chars.map(c => {
    const pos = D.graphPositions[c.id];
    const col = NODE_COLORS[c.importance||0];
    const isSelected = UI.activeEntityId === c.id;
    const isConnecting = UI.graphConnecting === c.id;
    return `<g class="g-node" id="gn-${c.id}"
      onmousedown="graphNodeMouseDown(event,'${c.id}')"
      onclick="graphNodeClick('${c.id}')">
      <circle cx="${pos.x}" cy="${pos.y}" r="${isSelected?22:18}"
        fill="${col}" opacity="${isConnecting?0.5:0.85}"
        stroke="${isSelected?'#c8a96e':isConnecting?'#e0c88a':'#2a2a3e'}"
        stroke-width="${isSelected?3:1.5}"/>
      <text class="g-label" x="${pos.x}" y="${pos.y+4}" fill="${isSelected?'#fff':'#e2d8c4'}" font-size="${isSelected?'12':'11'}" font-weight="${isSelected?'600':'400'}">${esc((c.name||'?').slice(0,6))}</text>
    </g>`;
  }).join('');

  setupGraphDrag();
}

function highlightGraphNode(id) {
  // Re-render to show selection
  renderGraph();
}

function graphNodeClick(id) {
  if (UI.graphConnecting !== null) {
    if (UI.graphConnecting === id) { cancelConnecting(); return; }
    // Create relationship
    showRelationshipDialog(UI.graphConnecting, id);
    cancelConnecting();
    return;
  }
  UI.activeEntityId = id;
  UI.activeCategory = 'characters';
  renderEntityList();
  renderDetailPanel(id);
  renderGraph();
}

function selectRelationship(id) {
  // Show relationship info
  const r = D.relationships.find(x => x.id === id);
  if (!r) return;
  const s = D.characters.find(x => x.id === r.source);
  const t = D.characters.find(x => x.id === r.target);
  toast(`${s?.name||'?'} ↔ ${t?.name||'?'}: ${r.type}`, 'info');
}

function setupGraphDrag() {
  const nodes = graphNodes.querySelectorAll('.g-node');
  nodes.forEach(node => {
    node.addEventListener('mousedown', (e) => {
      if (UI.graphConnecting) return;
      e.preventDefault();
    });
  });
}

function graphNodeMouseDown(e, id) {
  if (UI.graphConnecting) return;
  e.stopPropagation();
  UI.graphDragging = id;
  const svg = graphSVG.getBoundingClientRect();
  UI.graphDragStart = { mouseX: e.clientX, mouseY: e.clientY, nodeX: D.graphPositions[id].x, nodeY: D.graphPositions[id].y };
}

document.addEventListener('mousemove', (e) => {
  if (!UI.graphDragging) return;
  const dx = e.clientX - UI.graphDragStart.mouseX;
  const dy = e.clientY - UI.graphDragStart.mouseY;
  D.graphPositions[UI.graphDragging] = {
    x: UI.graphDragStart.nodeX + dx,
    y: UI.graphDragStart.nodeY + dy
  };
  renderGraph();
});

document.addEventListener('mouseup', () => {
  if (UI.graphDragging) {
    saveData('graphPositions');
    UI.graphDragging = null;
  }
});

function startConnecting() {
  UI.graphConnecting = '_waiting';
  document.getElementById('btn-connect').style.display = 'none';
  document.getElementById('btn-cancel-connect').style.display = 'inline-flex';
  document.getElementById('graph-hint-text').textContent = 'Nhấp vào nhân vật nguồn...';
  document.body.classList.add('connecting');
  graphNodes.querySelectorAll('.g-node').forEach(n => {
    n.style.cursor = 'crosshair';
  });
}

function cancelConnecting() {
  UI.graphConnecting = null;
  document.getElementById('btn-connect').style.display = 'inline-flex';
  document.getElementById('btn-cancel-connect').style.display = 'none';
  document.getElementById('graph-hint-text').textContent = '';
  document.body.classList.remove('connecting');
  renderGraph();
}

  // Normal click
  UI.activeEntityId = id;
  UI.activeCategory = 'characters';

  renderGraph();

function autoLayout() {
  const chars = D.characters;
  if (!chars.length) return;
  const rect = graphSVG.getBoundingClientRect();
  const w = rect.width || 600, h = rect.height || 400;
  const cols = Math.ceil(Math.sqrt(chars.length));
  chars.forEach((c, i) => {
    const row = Math.floor(i / cols), col = i % cols;
    D.graphPositions[c.id] = {
      x: 80 + col * ((w - 160) / Math.max(cols-1, 1)),
      y: 80 + row * ((h - 160) / Math.max(Math.ceil(chars.length/cols)-1, 1))
    };
  });
  saveData('graphPositions');
  renderGraph();
  toast('Đã sắp xếp đồ thị');
}

function resetGraphView() {
  D.graphPositions = {};
  saveData('graphPositions');
  renderGraph();
}

function toggleBookChapter(chId) {
  const b = D.books.find(x => x.id === UI.activeBookId);
  if (!b) return;
  b.chapters = b.chapters || [];
  const idx = b.chapters.indexOf(chId);
  if (idx > -1) b.chapters.splice(idx, 1);
  else b.chapters.push(chId);
  saveData('books');
  renderBookDetail();
  renderBookList();
}

}

function showContextMenu(e, items) {
  const menu = document.getElementById('ctx-menu');
  menu.innerHTML = items.map(item => item.sep
    ? `<div class="ctx-sep"></div>`
    : `<div class="ctx-item ${item.danger?'danger':''}">${item.label}</div>`
  ).join('');
  const clickableItems = menu.querySelectorAll('.ctx-item');
  const actions = items.filter(i => !i.sep);
  clickableItems.forEach((el, i) => {
    el.addEventListener('click', () => { hideContextMenu(); actions[i].action(); });
  });
  menu.style.display = 'block';
  menu.style.left = Math.min(e.clientX, window.innerWidth - 180) + 'px';
  menu.style.top = Math.min(e.clientY, window.innerHeight - menu.offsetHeight - 10) + 'px';
}

});

  UI.folderOpen[folderId] = true;

  saveAll();
}

function init() {
  loadData();
  loadDemoData();
  runValidation();
  renderFolderTree();
  renderEntityList();
  renderBookList();

  // Init graph after DOM is ready
  setTimeout(() => {
    initGraph();
    renderGraph();
  }, 100);

  // Click outside cmd dropdown
  document.addEventListener('click', (e) => {
    if (!e.target.closest('#cmd-drop') && !e.target.closest('#editor')) hideCmdDrop();
  });

  // Handle graph resize
  window.addEventListener('resize', () => {
    if (graphSVG) {
      const rect = graphSVG.getBoundingClientRect();
      graphW = rect.width; graphH = rect.height;
    }
  });
}

window.addEventListener('DOMContentLoaded', init);
